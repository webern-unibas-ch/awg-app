import { NgModule } from '@angular/core';

@NgModule({

        EditionTkaLabelComponent,
        EditionTkaTableComponent,

        SourceEvaluationComponent,
        SourceListComponent,
        routedEditionReportComponents,
    ],
})
export class EditionReportModule {}
