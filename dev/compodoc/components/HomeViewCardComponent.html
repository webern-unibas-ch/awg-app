import { NgTemplateOutlet } from '@angular/common';
import { ChangeDetectionStrategy, Component, input } from '@angular/core';
import { RouterLink } from '@angular/router';

import { FaIconComponent } from '@fortawesome/angular-fontawesome';
import { faArrowRight } from '@fortawesome/free-solid-svg-icons';

import { HomeViewCard } from './models/home-view-card.model';

/**
 * The HomeViewCard component.
 *
 * It contains the home card
 * for the home view of the app.
 */
@Component({
    selector: 'awg-home-view-card',
    templateUrl: './home-view-card.component.html',
    styleUrls: ['./home-view-card.component.scss'],
    changeDetection: ChangeDetectionStrategy.OnPush,
    imports: [FaIconComponent, NgTemplateOutlet, RouterLink],
})
export class HomeViewCardComponent {
    /**
     * Input signal: cardData.
     *
     * It holds the data for the home view card.
     */
    cardData = input.required<HomeViewCard>();

    /**
     * Public variable: faArrowRight.
     *
     * It instantiates fontawesome's faArrowRight icon.
     */
    faArrowRight = faArrowRight;
}

<div class="awg-home-view-card card h-100 shadow text-center">
    @let cardData = this.cardData();

    <img [src]="cardData.imgSrc" class="card-img-top" [alt]="cardData.imgAlt" />
    <div class="card-body">
        <h5 class="card-title">{{ cardData.title }}</h5>
        <p class="card-text">{{ cardData.text }}</p>
    </div>
    <div class="card-footer">
        <ng-template #linkTemplate>
            <span><fa-icon class="me-2" [icon]="faArrowRight" />{{ cardData.linkText }}</span>
        </ng-template>

        @switch (cardData.link.type) {
            @case ('internal') {
                <a class="btn btn-info text-light stretched-link" [routerLink]="cardData.link.route"
                    ><ng-container *ngTemplateOutlet="linkTemplate"
                /></a>
            }
            @case ('external') {
                <a class="btn btn-info text-light stretched-link" [href]="cardData.link.href"
                    ><ng-container *ngTemplateOutlet="linkTemplate"
                /></a>
            }
        }
    </div>
</div>
